## 1.1 PostgreSQL 简介

PostgreSQL 是世界上最先进的开源关系型数据库，具备以下核心优势：

- **ACID 事务**：完整的事务支持，保证数据一致性
- **MVCC**：多版本并发控制，读写互不阻塞
- **可扩展性**：支持自定义类型、函数、操作符、索引方法
- **丰富的数据类型**：JSON/JSONB、数组、几何类型、全文搜索等
- **标准兼容**：高度兼容 SQL 标准

### 安装与连接

```bash
# Ubuntu/Debian
sudo apt update
sudo apt install postgresql postgresql-contrib

# macOS (Homebrew)
brew install postgresql@16
brew services start postgresql@16

# Docker（推荐用于开发）
docker run --name pg-dev -e POSTGRES_PASSWORD=mypassword -p 5432:5432 -d postgres:16

# 连接数据库
psql -U postgres -h localhost -p 5432
```

## 1.2 数据库与表的管理

### 创建与管理数据库

```sql
-- 创建数据库
CREATE DATABASE myapp
  WITH ENCODING = 'UTF8'
  LC_COLLATE = 'en_US.UTF-8'
  TEMPLATE = template0;

-- 删除数据库
DROP DATABASE IF EXISTS myapp;

-- 查看数据库信息
SELECT datname, pg_size_pretty(pg_database_size(datname))
FROM pg_database
WHERE datistemplate = false;
```

### Schema 管理

Schema 是数据库内的命名空间，用于组织表和其他对象。

```sql
-- 创建 schema
CREATE SCHEMA app;
CREATE SCHEMA IF NOT EXISTS api;

-- 设置搜索路径
SET search_path TO app, public;

-- 查看当前搜索路径
SHOW search_path;

-- 删除 schema
DROP SCHEMA app CASCADE;  -- CASCADE 会删除 schema 下所有对象
```

### 创建表

```sql
-- 基础建表
CREATE TABLE users (
  id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  email       TEXT UNIQUE NOT NULL,
  username    VARCHAR(50) NOT NULL,
  avatar_url  TEXT,
  bio         TEXT DEFAULT '',
  is_active   BOOLEAN DEFAULT true,
  created_at  TIMESTAMPTZ DEFAULT now(),
  updated_at  TIMESTAMPTZ DEFAULT now()
);

-- 带外键关联的表
CREATE TABLE posts (
  id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  author_id   BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  title       VARCHAR(255) NOT NULL,
  content     TEXT,
  slug        TEXT UNIQUE NOT NULL,
  status      TEXT DEFAULT 'draft' CHECK (status IN ('draft', 'published', 'archived')),
  tags        TEXT[] DEFAULT '{}',          -- 数组类型
  metadata    JSONB DEFAULT '{}',           -- JSONB 类型
  view_count  INTEGER DEFAULT 0,
  published_at TIMESTAMPTZ,
  created_at  TIMESTAMPTZ DEFAULT now(),
  updated_at  TIMESTAMPTZ DEFAULT now()
);

-- 多对多关系中间表
CREATE TABLE post_likes (
  user_id    BIGINT REFERENCES users(id) ON DELETE CASCADE,
  post_id    BIGINT REFERENCES posts(id) ON DELETE CASCADE,
  created_at TIMESTAMPTZ DEFAULT now(),
  PRIMARY KEY (user_id, post_id)
);

-- 评论表（支持嵌套评论）
CREATE TABLE comments (
  id         BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  post_id    BIGINT NOT NULL REFERENCES posts(id) ON DELETE CASCADE,
  user_id    BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  parent_id  BIGINT REFERENCES comments(id) ON DELETE CASCADE,
  content    TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT now()
);
```

### 修改表结构

```sql
-- 添加列
ALTER TABLE users ADD COLUMN phone VARCHAR(20);

-- 修改列类型
ALTER TABLE users ALTER COLUMN phone TYPE TEXT;

-- 设置/取消默认值
ALTER TABLE users ALTER COLUMN is_active SET DEFAULT true;
ALTER TABLE users ALTER COLUMN is_active DROP DEFAULT;

-- 设置/取消 NOT NULL
ALTER TABLE users ALTER COLUMN bio SET NOT NULL;
ALTER TABLE users ALTER COLUMN bio DROP NOT NULL;

-- 添加约束
ALTER TABLE users ADD CONSTRAINT email_format CHECK (email ~* '^[^@]+@[^@]+\.[^@]+$');

-- 重命名列
ALTER TABLE users RENAME COLUMN avatar_url TO profile_image_url;

-- 重命名表
ALTER TABLE users RENAME TO app_users;

-- 删除列
ALTER TABLE users DROP COLUMN IF EXISTS phone;
```

## 1.3 PostgreSQL 核心数据类型

### 数值类型

```sql
-- 整数
SMALLINT        -- 2 字节, -32768 ~ 32767
INTEGER         -- 4 字节, -2^31 ~ 2^31-1
BIGINT          -- 8 字节, -2^63 ~ 2^63-1

-- 自增主键（推荐方式）
BIGINT GENERATED ALWAYS AS IDENTITY    -- 不允许手动插入
BIGINT GENERATED BY DEFAULT AS IDENTITY -- 允许手动插入

-- 浮点数
REAL            -- 4 字节浮点
DOUBLE PRECISION -- 8 字节浮点
NUMERIC(10, 2)  -- 精确数值, 适合金额计算

-- 序列（旧方式，不推荐）
SERIAL          -- 等同于 INTEGER + 序列
BIGSERIAL       -- 等同于 BIGINT + 序列
```

### 文本类型

```sql
TEXT            -- 变长文本, 无长度限制（推荐）
VARCHAR(n)      -- 变长文本, 最大 n 个字符
CHAR(n)         -- 定长文本, 不足补空格（少用）

-- TEXT vs VARCHAR: 在 PostgreSQL 中性能完全相同
-- 推荐统一使用 TEXT, 用 CHECK 约束控制长度
CREATE TABLE example (
  name TEXT CHECK (char_length(name) <= 100)
);
```

### 时间类型

```sql
DATE                    -- 日期: 2025-01-15
TIME                    -- 时间: 14:30:00
TIMESTAMP               -- 时间戳（无时区）
TIMESTAMPTZ             -- 时间戳（带时区, 推荐）
INTERVAL                -- 时间间隔: '1 year 2 months 3 days'

-- 时间操作示例
SELECT now();                              -- 当前时间戳
SELECT now() - INTERVAL '7 days';          -- 7天前
SELECT now()::date;                        -- 转为日期
SELECT date_trunc('month', now());         -- 截断到月初
SELECT extract(YEAR FROM now());           -- 提取年份
SELECT age(now(), '1990-01-01'::date);     -- 计算年龄
```

### JSON / JSONB 类型

```sql
-- JSONB 是二进制存储, 支持索引, 推荐使用
-- JSON 是文本存储, 保留格式但不支持索引

CREATE TABLE products (
  id       BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name     TEXT NOT NULL,
  attrs    JSONB DEFAULT '{}'
);

-- 插入 JSON 数据
INSERT INTO products (name, attrs) VALUES
  ('iPhone', '{"color": "black", "storage": 256, "specs": {"cpu": "A17", "ram": 8}}'),
  ('MacBook', '{"color": "silver", "storage": 512, "ports": ["usb-c", "hdmi"]}');

-- 查询 JSON 字段
SELECT name, attrs->>'color' AS color FROM products;                    -- 提取文本
SELECT name, attrs->'specs'->>'cpu' AS cpu FROM products;               -- 嵌套提取
SELECT name FROM products WHERE attrs->>'color' = 'black';             -- 条件过滤
SELECT name FROM products WHERE (attrs->>'storage')::int > 300;        -- 类型转换比较
SELECT name FROM products WHERE attrs ? 'ports';                       -- 键是否存在
SELECT name FROM products WHERE attrs @> '{"color": "black"}';         -- 包含判断
SELECT name FROM products WHERE attrs->'specs' @> '{"ram": 8}';       -- 嵌套包含

-- 更新 JSON 字段
UPDATE products SET attrs = attrs || '{"weight": "200g"}' WHERE name = 'iPhone';      -- 合并
UPDATE products SET attrs = attrs - 'weight' WHERE name = 'iPhone';                    -- 删除键
UPDATE products SET attrs = jsonb_set(attrs, '{specs,ram}', '16') WHERE name = 'iPhone'; -- 嵌套更新
```

### 数组类型

```sql
CREATE TABLE articles (
  id   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  title TEXT NOT NULL,
  tags  TEXT[] DEFAULT '{}'
);

-- 插入数组
INSERT INTO articles (title, tags) VALUES
  ('PostgreSQL Guide', ARRAY['database', 'postgresql', 'tutorial']),
  ('React Tips', '{"react", "frontend", "javascript"}');

-- 查询数组
SELECT title FROM articles WHERE 'postgresql' = ANY(tags);             -- 包含某元素
SELECT title FROM articles WHERE tags @> ARRAY['database'];            -- 包含子数组
SELECT title FROM articles WHERE tags && ARRAY['react', 'vue'];        -- 有交集
SELECT title, array_length(tags, 1) AS tag_count FROM articles;        -- 数组长度
SELECT title, unnest(tags) AS tag FROM articles;                        -- 展开数组

-- 更新数组
UPDATE articles SET tags = array_append(tags, 'new-tag') WHERE id = 1;   -- 追加
UPDATE articles SET tags = array_remove(tags, 'tutorial') WHERE id = 1;  -- 移除
```

### UUID 类型

```sql
-- PostgreSQL 16+ 内置 gen_random_uuid()
CREATE TABLE sessions (
  id         UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id    BIGINT REFERENCES users(id),
  token      TEXT NOT NULL,
  expires_at TIMESTAMPTZ NOT NULL
);

-- 旧版需要启用扩展
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
SELECT uuid_generate_v4();
```

### 枚举类型

```sql
-- 创建枚举
CREATE TYPE order_status AS ENUM ('pending', 'processing', 'shipped', 'delivered', 'cancelled');

CREATE TABLE orders (
  id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  status order_status DEFAULT 'pending'
);

-- 添加枚举值
ALTER TYPE order_status ADD VALUE 'refunded' AFTER 'cancelled';

-- 注意: 枚举值不能删除或重命名, 需要谨慎设计
-- 替代方案: 使用 TEXT + CHECK 约束（更灵活）
CREATE TABLE orders_v2 (
  id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  status TEXT DEFAULT 'pending' CHECK (status IN ('pending', 'processing', 'shipped', 'delivered', 'cancelled'))
);
```

## 1.4 CRUD 操作

### INSERT

```sql
-- 单条插入
INSERT INTO users (email, username)
VALUES ('alice@example.com', 'alice');

-- 多条插入
INSERT INTO users (email, username) VALUES
  ('bob@example.com', 'bob'),
  ('charlie@example.com', 'charlie');

-- 插入并返回数据
INSERT INTO users (email, username)
VALUES ('dave@example.com', 'dave')
RETURNING id, email, created_at;

-- 冲突处理 (UPSERT)
INSERT INTO users (email, username)
VALUES ('alice@example.com', 'alice_new')
ON CONFLICT (email) DO UPDATE
SET username = EXCLUDED.username,
    updated_at = now();

-- 冲突时忽略
INSERT INTO users (email, username)
VALUES ('alice@example.com', 'alice')
ON CONFLICT (email) DO NOTHING;

-- 从查询结果插入
INSERT INTO archived_posts (id, title, content)
SELECT id, title, content
FROM posts
WHERE status = 'archived';
```

### SELECT

```sql
-- 基本查询
SELECT * FROM users WHERE is_active = true;

-- 选择特定列并起别名
SELECT
  id,
  username AS name,
  email,
  extract(YEAR FROM age(now(), created_at)) AS years_active
FROM users;

-- 排序与分页
SELECT * FROM posts
ORDER BY created_at DESC
LIMIT 20 OFFSET 40;    -- 第3页, 每页20条

-- 高效分页（游标分页, 推荐用于大数据集）
SELECT * FROM posts
WHERE created_at < '2025-01-01'  -- 上一页最后一条的时间
ORDER BY created_at DESC
LIMIT 20;

-- DISTINCT
SELECT DISTINCT ON (author_id)
  id, author_id, title, created_at
FROM posts
ORDER BY author_id, created_at DESC;  -- 每个作者最新的文章

-- 条件过滤
SELECT * FROM posts
WHERE status = 'published'
  AND tags @> ARRAY['postgresql']
  AND created_at > now() - INTERVAL '30 days'
  AND title ILIKE '%guide%';           -- 不区分大小写模糊匹配

-- CASE 表达式
SELECT
  title,
  CASE
    WHEN view_count > 10000 THEN 'hot'
    WHEN view_count > 1000 THEN 'popular'
    ELSE 'normal'
  END AS popularity
FROM posts;

-- COALESCE (空值处理)
SELECT
  username,
  COALESCE(bio, '这个人很懒, 什么都没留下') AS bio
FROM users;

-- NULLIF
SELECT
  title,
  view_count / NULLIF(like_count, 0) AS views_per_like  -- 避免除零错误
FROM posts;
```

### UPDATE

```sql
-- 基本更新
UPDATE users
SET username = 'alice_updated', updated_at = now()
WHERE id = 1;

-- 带返回值的更新
UPDATE posts
SET status = 'published', published_at = now()
WHERE id = 5
RETURNING *;

-- 基于子查询的更新
UPDATE posts
SET view_count = view_count + 1
WHERE id IN (SELECT post_id FROM post_likes WHERE user_id = 1);

-- 使用 FROM 子句跨表更新
UPDATE posts p
SET status = 'archived'
FROM users u
WHERE p.author_id = u.id
  AND u.is_active = false;
```

### DELETE

```sql
-- 基本删除
DELETE FROM posts WHERE status = 'draft' AND created_at < now() - INTERVAL '1 year';

-- 带返回值的删除
DELETE FROM comments WHERE id = 10 RETURNING *;

-- 清空表（比 DELETE 快, 不触发触发器）
TRUNCATE TABLE logs;
TRUNCATE TABLE logs RESTART IDENTITY CASCADE;
```

## 1.5 约束

```sql
-- 主键约束
PRIMARY KEY

-- 唯一约束
UNIQUE
-- 多列唯一
UNIQUE (col1, col2)

-- 非空约束
NOT NULL

-- 外键约束
REFERENCES other_table(id) ON DELETE CASCADE
-- ON DELETE 选项:
--   CASCADE    -- 级联删除
--   SET NULL   -- 设为 NULL
--   SET DEFAULT -- 设为默认值
--   RESTRICT   -- 禁止删除（默认）

-- CHECK 约束
CHECK (price > 0)
CHECK (start_date < end_date)
CHECK (status IN ('active', 'inactive'))

-- 排除约束（防止范围重叠）
CREATE EXTENSION IF NOT EXISTS btree_gist;
CREATE TABLE reservations (
  id       BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  room_id  INTEGER NOT NULL,
  during   TSTZRANGE NOT NULL,
  EXCLUDE USING GIST (room_id WITH =, during WITH &&)
);
```

## 1.6 常用内置函数

### 字符串函数

```sql
SELECT length('hello');                    -- 5
SELECT upper('hello');                     -- HELLO
SELECT lower('HELLO');                     -- hello
SELECT trim('  hello  ');                  -- 'hello'
SELECT substring('hello world' FROM 1 FOR 5);  -- hello
SELECT replace('hello world', 'world', 'pg');  -- hello pg
SELECT concat('hello', ' ', 'world');      -- hello world
SELECT concat_ws(', ', 'a', 'b', 'c');    -- a, b, c
SELECT split_part('a.b.c', '.', 2);       -- b
SELECT regexp_replace('abc123', '[0-9]', 'X', 'g');  -- abcXXX
SELECT left('hello', 3);                  -- hel
SELECT right('hello', 3);                 -- llo
SELECT repeat('ab', 3);                   -- ababab
SELECT reverse('hello');                  -- olleh
```

### 聚合函数

```sql
SELECT
  count(*) AS total,
  count(DISTINCT author_id) AS unique_authors,
  sum(view_count) AS total_views,
  avg(view_count)::numeric(10,2) AS avg_views,
  max(view_count) AS max_views,
  min(view_count) AS min_views
FROM posts
WHERE status = 'published';

-- 数组聚合
SELECT author_id, array_agg(title ORDER BY created_at DESC) AS post_titles
FROM posts
GROUP BY author_id;

-- 字符串聚合
SELECT author_id, string_agg(title, ', ' ORDER BY created_at) AS post_list
FROM posts
GROUP BY author_id;

-- JSON 聚合
SELECT author_id, jsonb_agg(jsonb_build_object('id', id, 'title', title)) AS posts_json
FROM posts
GROUP BY author_id;

-- 条件聚合
SELECT
  count(*) FILTER (WHERE status = 'published') AS published_count,
  count(*) FILTER (WHERE status = 'draft') AS draft_count,
  avg(view_count) FILTER (WHERE status = 'published') AS avg_published_views
FROM posts;
```

### 日期与时间函数

```sql
SELECT now();                                          -- 当前时间戳
SELECT current_date;                                   -- 当前日期
SELECT current_time;                                   -- 当前时间
SELECT date_trunc('month', now());                     -- 2025-01-01 00:00:00+00
SELECT date_trunc('week', now());                      -- 本周一 00:00:00
SELECT extract(DOW FROM now());                        -- 0(周日)~6(周六)
SELECT extract(EPOCH FROM now());                      -- Unix 时间戳
SELECT to_char(now(), 'YYYY-MM-DD HH24:MI:SS');       -- 格式化
SELECT '2025-01-15'::date + INTERVAL '30 days';       -- 日期运算
SELECT generate_series(                                 -- 生成日期序列
  '2025-01-01'::date,
  '2025-12-31'::date,
  '1 month'::interval
);
```
